package tests

import (
	"net/http"
	"net/http/httptest"
	"testing"
	"time"

	"github.com/agiledragon/gomonkey/v2"
)

func TestFakesOnlyClockAndServer(t *testing.T) {
	patches := gomonkey.ApplyFunc(time.Now, func() time.Time { return time.Unix(0, 0) })
	defer patches.Reset()
	server := httptest.NewServer(http.NotFoundHandler())
	defer server.Close()
}
