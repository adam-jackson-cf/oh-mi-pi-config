package tests

import (
	"strconv"
	"testing"
)

func TestAdditionPreservesTotal(t *testing.T) {
	for _, value := range []int{1, 2} {
		t.Run(strconv.Itoa(value), func(t *testing.T) {
			if value+0 != value {
				t.Fatal(value)
			}
		})
	}
}

func TestMultiplicationPreservesIdentity(t *testing.T) {
	if 3*1 != 3 {
		t.Fatal("identity")
	}
}
